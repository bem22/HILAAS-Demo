cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(hilaas_zephyr_selftest)
target_sources(app PRIVATE main.c ../common/selftest.c)
target_include_directories(app PRIVATE ../common)
foreach(identity HILAAS_COMMIT HILAAS_RUN_ID HILAAS_NONCE HILAAS_BOARD)
  if(NOT DEFINED ${identity})
    message(FATAL_ERROR "Missing ${identity}")
  endif()
  target_compile_definitions(app PRIVATE ${identity}="${${identity}}")
endforeach()
